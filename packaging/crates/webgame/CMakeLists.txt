project(traineros_webgame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)

target_link_libraries(traineros-webgame PRIVATE Qt6::Quick Qt6::Qml Qt6::WebEngineQuick SDL2::SDL2)
